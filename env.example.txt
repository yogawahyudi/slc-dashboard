# =================================================================
# Database
# =================================================================
# PostgreSQL connection string. Needed at runtime and by the db:* scripts
# (not by `npm run build`).

DATABASE_URL=    # Example: "postgres://slc:password@localhost:5432/slc_polri"

# =================================================================
# Auth
# =================================================================
# Session cookies are marked Secure in production, so browsers only send them
# over HTTPS. Set to 'false' ONLY if production is served over plain HTTP.

SESSION_COOKIE_SECURE=    # Example: "false"

# Development only: skip sign-in so the dashboard runs without a database.
# Everyone is treated as an admin. Ignored by production builds.

AUTH_DISABLED=    # Example: "true"

# Fixed accounts without a database (works in production too). When set, sign-in
# uses only these accounts and DATABASE_URL is not used for auth.
# Format: "nrp:name:role:password", accounts separated by ';'
#   - nrp: 8 digits (or 18-digit NIP); role: admin | operator | viewer
#   - password: at least 10 characters, may contain ':' but not ';'.
#     Avoid '$' in .env files (it is expanded as a variable) or write it as '\$'.
# Change a password here and restart/redeploy: that account's sessions end.
# AUTH_SECRET (at least 32 characters) signs the session cookie; generate one with
#   node -e "console.log(require('crypto').randomBytes(32).toString('base64url'))"
# Changing AUTH_SECRET signs everyone out.

AUTH_USERS=     # Example: "12345678:Admin SLC:admin:GantiPassword1;87654321:Operator 1:operator:GantiPassword2"
AUTH_SECRET=

# Used once by `npm run db:seed-admin` to create the first admin.
# Leave ADMIN_PASSWORD empty to generate a temporary password.

ADMIN_NRP=         # Example: "12345678"
ADMIN_NAME=        # Example: "Admin SLC"
ADMIN_PASSWORD=

# =================================================================
# Build Configuration
# =================================================================
# Set to 'true' when deploying with Docker or self-hosting on a VPS.
# This enables Next.js standalone output mode for smaller deployments.

BUILD_STANDALONE=    # Example: "true"
